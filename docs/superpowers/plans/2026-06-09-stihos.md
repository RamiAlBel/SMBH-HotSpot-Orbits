```python
import torch
import torch.nn as nn


class RegressionHead(nn.Module):
    def __init__(
        self,
        input_dim: int,
        hidden_dims: tuple[int, int] = (256, 256),
        num_blocks: int = 2,
        dropout: float = 0.1
    ):
        super().__init__()
        layers: list[nn.Module] = []

        layers.append(nn.Linear(input_dim, hidden_dims[0]))
        layers.append(nn.BatchNorm1d(hidden_dims[0]))
        layers.append(nn.ReLU())
        layers.append(nn.Dropout(dropout))

        for _ in range(num_blocks):
            block = nn.Sequential(
                nn.Linear(hidden_dims[0], hidden_dims[0]),
                nn.BatchNorm1d(hidden_dims[0]),
                nn.ReLU(),
                nn.Dropout(dropout),
                nn.Linear(hidden_dims[0], hidden_dims[0]),
                nn.BatchNorm1d(hidden_dims[0]),
            )
            layers.append(block)

        self.stem = nn.Sequential(*layers)
        self.head = nn.Linear(hidden_dims[0], 1)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        out = self.stem[0](x)
        for layer in self.stem[1:]:
            if isinstance(layer, nn.Sequential):
                res = out
                out = layer(out)
                out = torch.relu(out + res)
            else:
                out = layer(out)
        return self.head(out)
```

```python
import numpy as np


def to_canonical(r, T, dpa, r_unit="M", T_unit="min", dpa_unit="deg"):
    """Convert inputs to canonical units: r[M], T[min], dpa[deg or list[deg]]."""
    if r_unit != "M":
        raise ValueError(f"Unknown r_unit: {r_unit!r}. Supported: 'M'.")

    if T_unit == "hours":
        T = float(T) * 60.0
    elif T_unit == "min":
        T = float(T)
    else:
        raise ValueError(f"Unknown T_unit: {T_unit!r}. Supported: 'min', 'hours'.")

    if dpa_unit == "rad":
        dpa = np.rad2deg(dpa)
    elif dpa_unit == "deg":
        pass
    else:
        raise ValueError(f"Unknown dpa_unit: {dpa_unit!r}. Supported: 'deg', 'rad'.")

    return float(r), T, dpa
```

```python
from pathlib import Path
import numpy as np
import torch
from ._model import RegressionHead

_RADIAN_TARGETS = {"incl", "theta"}


def load_checkpoint(path):
    """Load a .pth file. Returns (model_in_eval_mode, checkpoint_dict)."""
    ck = torch.load(str(path), map_location="cpu", weights_only=False)
    input_dim = ck["scaler_X_mean"].shape[0]
    model = RegressionHead(input_dim=input_dim)
    model.load_state_dict(ck["model_state_dict"])
    model.eval()
    return model, ck


def mc_predict(model, ck, x_orig, n_mc, sigma_obs, target_name):
    """
    Run n_mc noisy forward passes and return raw prediction array in physical units.

    x_orig     : 1-D float array [r, T, dpa...] in canonical units (M, min, deg)
    sigma_obs  : 1-D float array, same length as x_orig (noise std per feature)
    target_name: "spin" | "incl" | "theta" | "z"
    """
    rng = np.random.default_rng()
    D = len(x_orig)
    x_noisy = x_orig[np.newaxis, :] + rng.standard_normal((n_mc, D)) * sigma_obs[np.newaxis, :]
    x_norm = (x_noisy - ck["scaler_X_mean"]) / ck["scaler_X_scale"]
    tensor = torch.tensor(x_norm, dtype=torch.float32)
    with torch.no_grad():
        y_norm = model(tensor).squeeze().numpy()
    y = y_norm * ck["scaler_y_scale"][0] + ck["scaler_y_mean"][0]
    if target_name in _RADIAN_TARGETS:
        y = np.rad2deg(y)
    return y


def predict_target(checkpoint_dir, target_name, x_orig, sigma_obs, n_mc):
    """
    Load all seed checkpoints for one target, run MC, pool samples.
    Returns (mean: float, std: float) in physical units.
    """
    target_dir = Path(checkpoint_dir) / target_name
    pth_files = sorted(target_dir.glob("model_seed*.pth"))
    if not pth_files:
        raise FileNotFoundError(f"No checkpoints found in {target_dir}")

    all_samples = []
    for pth in pth_files:
        model, ck = load_checkpoint(pth)
        samples = mc_predict(model, ck, x_orig, n_mc, sigma_obs, target_name)
        all_samples.append(samples)

    pooled = np.concatenate(all_samples)
    return float(np.mean(pooled)), float(np.std(pooled))
```

```python
_CHECKPOINT_ROOT = Path(__file__).parent / "checkpoints"


@dataclass
class ParameterEstimate:
    mean: float
    std: float

    def __repr__(self):
        return f"{self.mean:.4f} ± {self.std:.4f}"


@dataclass
class PredictionResult:
    spin: ParameterEstimate
    incl: ParameterEstimate
    theta: Optional[ParameterEstimate] = None
    z: Optional[ParameterEstimate] = None

    def to_dict(self):
        d = {
            "spin": {"mean": self.spin.mean, "std": self.spin.std},
            "incl": {"mean": self.incl.mean, "std": self.incl.std},
        }
        if self.theta is not None:
            d["theta"] = {"mean": self.theta.mean, "std": self.theta.std}
        if self.z is not None:
            d["z"] = {"mean": self.z.mean, "std": self.z.std}
        return d

    def __str__(self):
        lines = [
            f" {'Parameter':<15} {'Mean':>10}  {'± Std':>10}",
            " " + "─" * 37,
            f" {'spin α':<15} {self.spin.mean:>10.4f}  {self.spin.std:>10.4f}",
            f" {'incl i (°)':<15} {self.incl.mean:>10.3f}  {self.incl.std:>10.3f}",
        ]
        if self.theta is not None:
            lines.append(
                f" {'theta (°)':<15} {self.theta.mean:>10.3f}  {self.theta.std:>10.3f}"
            )
        if self.z is not None:
            lines.append(
                f" {'z (M)':<15} {self.z.mean:>10.4f}  {self.z.std:>10.4f}"
            )
        return "\n".join(lines)


def predict(
    r,
    T,
    dpa,
    orbit="equatorial",
    r_unit="M",
    T_unit="min",
    dpa_unit="deg",
    n_mc=2000,
):
    """
    Predict SMBH parameters from a single hotspot observation.

    Parameters
    ----------
    r      : float — hotspot orbital radius
    T      : float — hotspot orbital period
    dpa    : float or list[float]
             Single float → avg mode (uses eq_avg models, equatorial only).
             List of exactly 10 floats → timeseries mode.
    orbit  : "equatorial" or "nonequatorial" (aliases: "eq", "noneq")
    r_unit : "M" (default)
    T_unit : "min" (default) or "hours"
    dpa_unit: "deg" (default) or "rad"
    n_mc   : int — MC noisy forward passes per seed checkpoint (default 2000)

    Returns
    -------
    PredictionResult
    """
    r_c, T_c, dpa_c = to_canonical(r, T, dpa, r_unit=r_unit, T_unit=T_unit, dpa_unit=dpa_unit)

    dpa_arr = np.atleast_1d(np.asarray(dpa_c, dtype=np.float64)).ravel()

    if dpa_arr.size == 1:
        exp_dir = _CHECKPOINT_ROOT / "eq_avg"
        x = np.array([r_c, T_c, float(dpa_arr[0])], dtype=np.float64)
    elif dpa_arr.size == 10:
        orbit_lower = orbit.lower()
        if orbit_lower in ("equatorial", "eq"):
            exp_dir = _CHECKPOINT_ROOT / "eq_full"
        elif orbit_lower in ("nonequatorial", "noneq", "non-equatorial"):
            exp_dir = _CHECKPOINT_ROOT / "noneq_full"
        else:
            raise ValueError(
                f"Unknown orbit: {orbit!r}. Use 'equatorial' or 'nonequatorial'."
            )
        x = np.concatenate([[r_c, T_c], dpa_arr])
    else:
        raise ValueError(
            f"dpa must be a single float (avg mode) or a list of exactly 10 values "
            f"(timeseries mode), got {dpa_arr.size} values."
        )

    sigma_obs = np.array([0.1, 2.0] + [5.0] * (len(x) - 2))

    spin_mean, spin_std = predict_target(exp_dir, "spin", x, sigma_obs, n_mc)
    incl_mean, incl_std = predict_target(exp_dir, "incl", x, sigma_obs, n_mc)

    result = PredictionResult(
        spin=ParameterEstimate(spin_mean, spin_std),
        incl=ParameterEstimate(incl_mean, incl_std),
    )

    if orbit.lower() in ("nonequatorial", "noneq", "non-equatorial"):
        theta_mean, theta_std = predict_target(exp_dir, "theta", x, sigma_obs, n_mc)
        z_mean, z_std = predict_target(exp_dir, "z", x, sigma_obs, n_mc)
        result.theta = ParameterEstimate(theta_mean, theta_std)
        result.z = ParameterEstimate(z_mean, z_std)

    return result
```

```python
import argparse
import sys
from . import predict


def main():
    parser = argparse.ArgumentParser(
        prog="stihos",
        description="SpaceTime Inference via Hotspot ObservationS — predict SMBH parameters",
    )
    parser.add_argument("--r", type=float, required=True,
                        help="Hotspot orbital radius")
    parser.add_argument("--T", type=float, required=True,
                        help="Hotspot orbital period")
    parser.add_argument("--dpa", type=float, nargs="+", required=True,
                        help="DPA angle(s): single value for avg mode, "
                             "10 values for timeseries mode")
    parser.add_argument("--orbit",
                        choices=["equatorial", "eq", "nonequatorial", "noneq"],
                        default="equatorial",
                        help="Orbit assumption (default: equatorial)")
    parser.add_argument("--r-unit", default="M",
                        help="Unit for r (default: M)")
    parser.add_argument("--T-unit", dest="T_unit", default="min",
                        choices=["min", "hours"],
                        help="Unit for T (default: min)")
    parser.add_argument("--dpa-unit", default="deg",
                        choices=["deg", "rad"],
                        help="Unit for DPA (default: deg)")
    parser.add_argument("--n-mc", type=int, default=2000,
                        help="MC forward passes per seed (default: 2000)")

    args = parser.parse_args()

    dpa_input = args.dpa[0] if len(args.dpa) == 1 else args.dpa

    try:
        result = predict(
            r=args.r,
            T=args.T,
            dpa=dpa_input,
            orbit=args.orbit,
            r_unit=args.r_unit,
            T_unit=args.T_unit,
            dpa_unit=args.dpa_unit,
            n_mc=args.n_mc,
        )
    except ValueError as e:
        print(f"Error: {e}", file=sys.stderr)
        sys.exit(1)

    orbit_label = (
        "equatorial" if args.orbit in ("equatorial", "eq") else "non-equatorial"
    )
    mode = "avg DPA" if isinstance(dpa_input, float) else f"timeseries ({len(dpa_input)} pts)"

    print()
    print("STIHOS — SpaceTime Inference via Hotspot ObservationS")
    print(f"Orbit: {orbit_label} | Mode: {mode} | n_mc: {args.n_mc} (×3 seeds)")
    print()
    print(result)
    print()


if __name__ == "__main__":
    main()
```
